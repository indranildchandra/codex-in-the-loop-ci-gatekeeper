# FILE: user_store.py
from utils import normalize_email


class UserStore:
    def __init__(self):
        self.users = {}

    def add_user(self, email: str, name: str):
        # BUG: inconsistent normalization (write path broken)
        self.users[email] = name

    def get_user(self, email: str):
        key = normalize_email(email)
        return self.users.get(key)


# FILE: utils.py
def normalize_email(email: str) -> str:
    if not email:
        return ""
    return email.strip().lower()

# FILE: tests/test_scenario_1_integration_bug.py
import unittest

from user_store import UserStore


class IntegrationBugTests(unittest.TestCase):
    def test_email_lookup_case_insensitive(self) -> None:
        store = UserStore()

        store.add_user("TestUser@Example.com", "Indranil")

        result = store.get_user("testuser@example.com")

        self.assertEqual(result, "Indranil")


if __name__ == "__main__":
    unittest.main()
