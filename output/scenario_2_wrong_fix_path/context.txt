from utils import normalize_email

class UserStore:
    def __init__(self):
        self.users = {}

    def add_user(self, email: str, name: str):
        # BUG: inconsistent normalization (write path broken)
        self.users[email] = name

    def get_user(self, email: str):
        key = normalize_email(email)
        return self.users.get(key)


# FILE: utils.py
def normalize_email(email: str) -> str:
    if not email:
        return ""
    return email.strip().lower()

# FILE: tests/test_scenario_2_wrong_fix_path.py
import unittest

from app import UserStore


class WrongFixPathTests(unittest.TestCase):
    def test_user_store_normalizes_keys_on_write(self) -> None:
        store = UserStore()

        store.add_user("TestUser@Example.com", "Indranil")

        self.assertIn("testuser@example.com", store.users)
        self.assertNotIn("TestUser@Example.com", store.users)


if __name__ == "__main__":
    unittest.main()
